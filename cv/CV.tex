% Section headings, set small-caps in the left margin.
%
% The heading sits in \smash, so it has zero height and the page builder
% cannot see it. Without \Needspace a heading can be left stranded at the
% foot of a page with its content on the next one -- and, worse, a
% three-line margin heading can overprint the footer. Reserve room for the
% heading plus the first few entries.

    \Needspace*{5\baselineskip}%
    \vspace{1.3\baselineskip}%
    \phantomsection\addcontentsline{toc}{section}{#1}%
    \noindent\llap{\scshape\smash{\parbox[t]{\marginparwidth}{\raggedright #1}}}%
    \vspace{-\baselineskip}\par}
